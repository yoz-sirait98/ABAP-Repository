form get_batch_master_config.

*Material which have batch"
  select matnr werks charg bwtar
  into corresponding fields of ibatch
  from mcha where
     matnr in  matnr and
     werks in werks.
    append ibatch.
  endselect.

*Material which have no batch"
  select matnr werks lgort labst lvorm into corresponding fields of imard
                       from mard
                       where matnr in matnr and
                             werks in werks.
    read table ibatch with key matnr = imard-matnr
                              werks = imard-werks.
    check sy-subrc ne 0.
    append imard.
  endselect.

*Read price, Tot QTY and else"
  select matnr bwkey bwtar lbkum salk3 vprsv verpr stprs lvorm
  into corresponding fields of idata
  from mbew
  where matnr in matnr and bwkey in werks.
    read table ibatch with key matnr = idata-matnr
                               werks = idata-bwkey.
    check sy-subrc ne 0.
    append idata.
  endselect.

endform.                    " GET_BATCH_MASTER_CONFIG

*&      Form  FURTHER_CALCULATION
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
form further_calculation.
*material with batch"
  loop at ibatch into ibatchwa.
    clear iresult.
    move-corresponding ibatchwa to iresult.
    select lgort clabs into corresponding fields of iresult
    from mchb where matnr eq ibatchwa-matnr and
                    werks eq ibatchwa-werks and
                    charg eq ibatchwa-charg.
      select single vprsv verpr stprs salk3 lbkum lvorm
             into corresponding fields of iresult from mbew
             where matnr eq ibatchwa-matnr and bwkey eq ibatchwa-werks
             and bwtar eq space. "icatchwa-bwtar.
      if iresult-vprsv eq 'V'.
        if iresult-lbkum ne 0.
          iresult-value = iresult-clabs *
                          ( iresult-salk3 / iresult-lbkum )  * 100.
        else.
          iresult-value = 0.
        endif.
      else.
        iresult-value = iresult-clabs * iresult-stprs * 100.
      endif.
      if iresult-sobkz = 'K'.
        if iresult-clabs <> 0.
          if iresult-value = 0.
            append iresult.
          endif.
        endif.
      else.
        append iresult.
      endif.
      if stk_cnsg = 'X'.
*    SELECT matnr werks lgort charg sobkz lifnr slabs
        select sobkz slabs into (iresult-sobkz, iresult-clabs )
        from mkol
        where matnr eq ibatchwa-matnr and
                        werks eq ibatchwa-werks and
                        lgort = iresult-lgort and
                        charg eq ibatchwa-charg.
          iresult-value = 0.
          append iresult.
        endselect.
      endif.
    endselect.
  endloop.

*material which have no batch"
  data: dumm type c, zverpr(12) type p decimals 7. " like mbew-verpr.
  loop at imard.
    clear iresult.
*  IF IDATA-MATNR EQ '000000000020000299'. "for debug"
*  MOVE 'X' TO DUMM.
*  ENDIF.
    move: imard-matnr to iresult-matnr,
          imard-werks to iresult-werks,
          imard-lgort to iresult-lgort,
          imard-lvorm to iresult-lvorm.

    read table idata with key matnr = iresult-matnr
                              bwkey = iresult-werks.
    if sy-subrc eq 0.
      move: idata-vprsv to iresult-vprsv,
            idata-verpr to iresult-verpr,
            idata-stprs to iresult-stprs.

      if imard-labst < idata-lbkum.
        move: imard-labst to iresult-clabs.
        if iresult-vprsv eq 'V'.
          if idata-lbkum ne 0.
            zverpr = ( idata-salk3 / idata-lbkum ).
          else.
            zverpr = 0.
          endif.
          iresult-value = iresult-clabs * zverpr * 100.
        else.
          iresult-value = iresult-clabs * iresult-stprs * 100.
        endif.
      else.
        move idata-lbkum to iresult-clabs.
        iresult-value = idata-salk3 * 100.
      endif.
* ---> Start of insertion ---> 04.03.2005
    else.
      move: imard-labst to iresult-clabs.
      iresult-value = iresult-clabs * iresult-stprs * 100.
* ---> End of insertion -----> 04.03.2005
    endif.
    append iresult.
    if stk_cnsg = 'X'.
*    SELECT matnr werks lgort charg sobkz lifnr slabs
      select sobkz slabs into (iresult-sobkz, iresult-clabs )
      from mkol
      where matnr eq imard-matnr and
                      werks eq imard-werks and
                      lgort = imard-lgort and
                      charg = ' '.
        iresult-value = 0.
        append iresult.
      endselect.
    endif.
  endloop.

endform.                    " FURTHER_CALCULATION

form check_authorization.
  data: l_werks like t001w-werks.

  clear g_authchk.
  select werks into l_werks
    from t001w
    where werks in werks.
    authority-check object 'M_MSEG_WMB'
             id 'ACTVT' dummy
             id 'WERKS' field l_werks.
    if sy-subrc <> 0.
      g_authchk = sy-subrc.
      message s000(es) with 'Please check your authorization'.
      exit.
    endif.
  endselect.
endform.                    " check_authorization
